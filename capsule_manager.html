""" CAPSULE MANAGER — V1.0 Il cavo mancante tra Oracle Auto e il bot. RUOLO: Oracle inserisce SuperCapsule nella tabella 'capsule' del DB SQLite. Il bot chiama CapsuleManager.valuta(contesto) ad ogni tick. CapsuleManager legge dal DB, valuta i trigger, ritorna l'azione. INTERFACCIA ATTESA DAL BOT: CapsuleManager(db_path, asset) .valuta(ctx) → {blocca, reason, soglia_delta, size_mult, profit_lock_min} .get_stats() → {attive, auto, static, learned, scadono_presto, trade_osservati} .get_active_capsules() → lista capsule attive per dashboard TABELLA DB (creata da oracle_auto._apply_capsule): capsule (id, asset, livello, tipo, trigger_json, azione_json, priority, enabled, samples, wr, pnl_avg, created_ts, scade_ts, hits, hits_saved, note) TRIGGER FIELDS (nomi esatti usati dal bot e Oracle): momentum, volatility, trend, regime, direction, drift_pct, oi_stato AZIONI (nomi esatti dal prompt L2): BLOCCA_ENTRY → blocca completamente l'entry ALZA_SOGLIA → aggiunge N punti alla soglia ABBASSA_SOGLIA → sottrae N punti dalla soglia ALZA_SOGLIA_USCITA → imposta profit_lock_min per l'exit FLOOR DINAMICO: - Capsule STATIC (dati storici): floor 48 rispettato - Capsule AUTO (Oracle con evidenza reale): floor 40 - Capsule AUTO con priority >= 9 (pattern confermato): floor 35 Razionale: il floor 48 è per proteggere da capsule non calibrate. Oracle ha già verificato con Phantom e dati reali — può andare più in basso. RELOAD: Il DB viene riletto ogni 60s o ad ogni cambio (hash check). Zero deploy — Oracle inserisce, bot legge al prossimo tick. """ import sqlite3 import json import time import os import hashlib import logging import operator log = logging.getLogger(__name__) OPS = { '>': operator.gt, '>=': operator.ge, '<': operator.lt, '<=': operator.le, '==': operator.eq, '!=': operator.ne, 'in': lambda a, b: a in b, 'not_in': lambda a, b: a not in b, } class CapsuleManager: """ Legge capsule dal DB SQLite e le applica al contesto di trading. Sostituisce CapsuleRuntime + ConfigHotReloader + IntelligenzaAutonoma. Asset-aware. Zero deploy. Loop chiuso con Oracle. """ RELOAD_INTERVAL = 60 # secondi tra reload DB # Floor dinamico per livello capsule FLOOR_STATIC = 48 # capsule hardcoded storiche FLOOR_AUTO = 40 # capsule Oracle con evidenza reale FLOOR_ORACLE = 35 # capsule Oracle priority >= 9 (pattern confermato) def __init__(self, db_path: str, asset: str = "BTCUSDC"): self.db_path = db_path self.asset = asset self._capsule_file = "capsule_attive.json" # compatibilità legacy self._capsule_cache: list = [] # capsule attive in memoria self._last_reload = 0.0 self._last_db_hash = "" self._trade_count = 0 self._load_capsule() log.info(f"[CM] ✅ CapsuleManager inizializzato — asset={asset} " f"capsule={len(self._capsule_cache)}") # ═══════════════════════════════════════════════════════════════ # INTERFACCIA PUBBLICA # ═══════════════════════════════════════════════════════════════ def valuta(self, ctx: dict) -> dict: """ Valuta il contesto corrente contro tutte le capsule attive. Chiamato dal bot ad ogni tick prima dell'entry e nell'exit. ctx keys: momentum, volatility, trend, regime, direction, drift_pct, oi_stato, oi_carica Ritorna: { blocca: bool, reason: str, soglia_delta: float, # negativo = abbassa, positivo = alza size_mult: float, profit_lock_min: float, } """ # Reload se necessario self._check_reload() risultato = { 'blocca': False, 'reason': '', 'soglia_delta': 0.0, 'size_mult': 1.0, 'profit_lock_min': 0.0, } ora = time.time() # Ordina per priority desc — capsule più importanti prima capsule_ordinate = sorted( self._capsule_cache, key=lambda c: c.get('priority', 5), reverse=True ) for cap in capsule_ordinate: # Skip disabilitate o scadute if not cap.get('enabled', True): continue # Non saltiamo per scade_ts — il giudice è il Phantom, non il tempo # Una capsula vive finché il mercato non dimostra che è sbagliata # Valuta trigger triggers = cap.get('trigger_parsed', []) if triggers and not self._match_triggers(triggers, ctx): continue # Match! Applica azione azione = cap.get('azione_parsed', {}) tipo = azione.get('tipo', '') valore = azione.get('valore', 0) livello = cap.get('livello', 'STATIC') priority = cap.get('priority', 5) if tipo == 'BLOCCA_ENTRY': risultato['blocca'] = True risultato['reason'] = ( azione.get('motivo') or cap.get('note', '') or f"CM_{livello}_{cap.get('id','?')}" )[:80] # BLOCCA ha priorità assoluta — stop log.debug(f"[CM] BLOCCA: {cap.get('id')} ctx={ctx.get('momentum')}|" f"{ctx.get('volatility')}|{ctx.get('trend')}") break elif tipo == 'ALZA_SOGLIA': delta = float(valore) if valore else 0.0 risultato['soglia_delta'] += delta log.debug(f"[CM] ALZA_SOGLIA +{delta} da {cap.get('id')}") elif tipo == 'ABBASSA_SOGLIA': # Floor dinamico — dipende dal livello della capsule delta = -abs(float(valore)) if valore else 0.0 floor = self._get_floor(livello, priority) risultato['soglia_delta'] += delta risultato['_floor'] = floor # passa il floor al bot log.debug(f"[CM] ABBASSA_SOGLIA {delta} (floor={floor}) da {cap.get('id')}") elif tipo == 'ALZA_SOGLIA_USCITA': min_profit = float(valore) if valore else 2.5 risultato['profit_lock_min'] = max( risultato['profit_lock_min'], min_profit ) log.debug(f"[CM] PROFIT_LOCK min={min_profit} da {cap.get('id')}") # Registra hit self._registra_hit(cap.get('id', '')) return risultato def get_stats(self) -> dict: """Esposto al heartbeat per dashboard ia_stats.""" self._check_reload() ora = time.time() attive = [c for c in self._capsule_cache if c.get('enabled') and (not c.get('scade_ts') or c['scade_ts'] > ora)] static = [c for c in attive if c.get('livello') == 'STATIC'] auto = [c for c in attive if c.get('livello') == 'AUTO'] learned = [c for c in attive if c.get('livello') == 'L2'] presto = [c for c in attive if c.get('scade_ts') and 0 < c['scade_ts'] - ora < 300] return { 'attive': len(attive), 'static': len(static), 'auto': len(auto), 'learned': len(learned), 'scadono_presto': len(presto), 'trade_osservati': self._trade_count, } def get_active_capsules(self) -> list: """Per dashboard ia_capsule_attive.""" self._check_reload() ora = time.time() result = [] for c in self._capsule_cache: if not c.get('enabled', True): continue scade = c.get('scade_ts', 0) # Non filtriamo per scade_ts — il giudice è il Phantom ttl = 0 # non rilevante — vita = fino a prova contraria result.append({ 'id': c.get('id', '?'), 'tipo': c.get('tipo', '?'), 'livello': c.get('livello', '?'), 'ttl_seconds': ttl, 'enabled': True, 'fonte': 'CM_DB', 'motivo': c.get('note', '')[:60], 'priority': c.get('priority', 5), }) return result def registra_trade(self, trade: dict): """Chiamato al close di ogni trade — aggiorna samples e WR delle capsule.""" self._trade_count += 1 # Cleanup capsule scadute ogni 50 trade if self._trade_count % 50 == 0: self._cleanup_scadute() # ═══════════════════════════════════════════════════════════════ # RELOAD DB # ═══════════════════════════════════════════════════════════════ def _check_reload(self): """Ricarica dal DB se sono passati RELOAD_INTERVAL secondi.""" now = time.time() if now - self._last_reload >= self.RELOAD_INTERVAL: self._load_capsule() def _load_capsule(self): """Legge tutte le capsule abilitate dal DB.""" try: conn = sqlite3.connect(self.db_path, timeout=5) conn.row_factory = sqlite3.Row now = time.time() # Leggi capsule attive non scadute per questo asset rows = conn.execute(""" SELECT id, asset, livello, tipo, trigger_json, azione_json, priority, enabled, samples, wr, pnl_avg, created_ts, scade_ts, hits, note FROM capsule WHERE enabled = 1 AND (asset = ? OR asset = 'ALL') ORDER BY priority DESC """, (self.asset,)).fetchall() conn.close() capsule = [] for r in rows: try: triggers = json.loads(r['trigger_json'] or '[]') azione = json.loads(r['azione_json'] or '{}') except Exception: triggers, azione = [], {} cap = dict(r) cap['trigger_parsed'] = triggers cap['azione_parsed'] = azione capsule.append(cap) # Carica anche le capsule dal file JSON legacy (STATIC hardcoded) json_caps = self._load_json_capsule() # Unisci — DB ha priorità su ID identici db_ids = {c['id'] for c in capsule} for jc in json_caps: if jc.get('id') not in db_ids: capsule.append(jc) self._capsule_cache = capsule self._last_reload = time.time() log.info(f"[CM] Ricaricate {len(capsule)} capsule dal DB " f"({len(json_caps)} da JSON legacy)") except Exception as e: log.error(f"[CM] Errore reload DB: {e}") # Mantieni cache precedente self._last_reload = time.time() def _load_json_capsule(self) -> list: """Carica capsule legacy da capsule_attive.json (STATIC).""" try: if not os.path.exists(self._capsule_file): return [] with open(self._capsule_file) as f: raw = json.load(f) result = [] now = time.time() for c in raw: if not c.get('enabled', True): continue scade = c.get('scade_ts', 0) if scade and scade < now: continue # Normalizza formato cap = { 'id': c.get('capsule_id') or c.get('id', '?'), 'asset': c.get('asset', self.asset), 'livello': c.get('livello', 'STATIC'), 'tipo': c.get('tipo', 'VETO'), 'priority': c.get('priority', 5), 'enabled': True, 'scade_ts': scade or None, 'note': c.get('descrizione', '')[:80], 'trigger_parsed': c.get('trigger', []), 'azione_parsed': c.get('azione', {}), } result.append(cap) return result except Exception as e: log.debug(f"[CM] JSON legacy: {e}") return [] # ═══════════════════════════════════════════════════════════════ # TRIGGER MATCHING # ═══════════════════════════════════════════════════════════════ def _match_triggers(self, triggers: list, ctx: dict) -> bool: """Ritorna True se TUTTI i trigger matchano il contesto.""" for t in triggers: # Supporta sia formato Oracle {campo, op, valore} # che formato legacy {param, op, value} campo = t.get('campo') or t.get('param', '') op = t.get('op', '==') valore = t.get('valore') or t.get('value') if campo not in ctx: continue # campo non presente = skip (non blocca) if op not in OPS: continue try: if not OPS[op](ctx[campo], valore): return False except Exception: return False return True # ═══════════════════════════════════════════════════════════════ # FLOOR DINAMICO # ═══════════════════════════════════════════════════════════════ def _get_floor(self, livello: str, priority: int) -> float: """ Floor dinamico per ABBASSA_SOGLIA. Capsule Oracle con evidenza reale possono andare sotto 48. """ if livello == 'STATIC': return self.FLOOR_STATIC # 48 — storico senza evidenza live if livello == 'AUTO' and priority >= 9: return self.FLOOR_ORACLE # 35 — Oracle con pattern confermato if livello == 'AUTO': return self.FLOOR_AUTO # 40 — Oracle con evidenza reale return self.FLOOR_STATIC # default conservativo # ═══════════════════════════════════════════════════════════════ # HIT TRACKING # ═══════════════════════════════════════════════════════════════ def _registra_hit(self, cap_id: str): """Incrementa hits nel DB — asincrono, non bloccante.""" try: conn = sqlite3.connect(self.db_path, timeout=2) conn.execute( "UPDATE capsule SET hits = hits + 1 WHERE id = ?", (cap_id,) ) conn.commit() conn.close() except Exception: pass # non critico # ═══════════════════════════════════════════════════════════════ # CLEANUP # ═══════════════════════════════════════════════════════════════ def _cleanup_scadute(self): """ Il giudice è il Phantom — non il tempo. Una capsula BLOCCA_ENTRY muore solo quando: - wr_reale > 50% su samples >= 20 → il mercato è cambiato, il pattern non è più tossico Una capsula ABBASSA_SOGLIA muore quando: - ha portato a perdite (wr_reale < 20% su samples >= 10 con pnl_avg < -1.0) Una capsula non viene MAI rimossa solo per il tempo. Se il contesto non si ripresenta → la capsula dorme in attesa. Quando torna, è pronta. """ try: conn = sqlite3.connect(self.db_path, timeout=5) # RIMUOVI: capsule BLOCCA che bloccano vincenti — mercato cambiato # WR bloccati > 50% su 20+ samples = stava bloccando trade vincenti conn.execute(""" UPDATE capsule SET enabled = 0 WHERE livello = 'AUTO' AND samples >= 20 AND wr > 0.50 AND azione_json LIKE '%BLOCCA_ENTRY%' """) # RIMUOVI: capsule ABBASSA_SOGLIA che peggiorano il sistema # Se con quella capsula attiva il bot ha perso sistematicamente conn.execute(""" UPDATE capsule SET enabled = 0 WHERE livello = 'AUTO' AND samples >= 10 AND wr < 0.20 AND pnl_avg < -1.0 AND azione_json LIKE '%ABBASSA_SOGLIA%' """) conn.commit() conn.close() log.info(f"[CM] Cleanup Phantom-driven eseguito") except Exception as e: log.debug(f"[CM] Cleanup: {e}") # ═══════════════════════════════════════════════════════════════ # COMPATIBILITÀ LEGACY (CapsuleRuntime interface) # ═══════════════════════════════════════════════════════════════ def reload(self) -> bool: """Compatibilità CapsuleRuntime.""" old_n = len(self._capsule_cache) self._load_capsule() return len(self._capsule_cache) != old_n def check_reload(self) -> bool: """Compatibilità ConfigHotReloader.""" return self.reload() def force_reload(self) -> bool: """Compatibilità ConfigHotReloader.""" self._last_reload = 0 return self.reload() def analizza_e_genera(self) -> list: """Compatibilità IntelligenzaAutonoma — no-op, Oracle fa tutto.""" return [] def registra_trade_ia(self, trade: dict): """Compatibilità IntelligenzaAutonoma.""" self.registra_trade(trade) @property def capsule_file(self): return self._capsule_file # ═══════════════════════════════════════════════════════════════════════ # SETUP TABELLA — chiamato al primo avvio se tabella non esiste # ═══════════════════════════════════════════════════════════════════════ def ensure_capsule_table(db_path: str): """Crea la tabella capsule se non esiste.""" try: conn = sqlite3.connect(db_path, timeout=10) conn.execute(""" CREATE TABLE IF NOT EXISTS capsule ( id TEXT PRIMARY KEY, asset TEXT DEFAULT 'BTCUSDC', livello TEXT DEFAULT 'AUTO', tipo TEXT DEFAULT 'PARAMETRO', trigger_json TEXT DEFAULT '[]', azione_json TEXT DEFAULT '{}', priority INTEGER DEFAULT 7, enabled INTEGER DEFAULT 1, samples INTEGER DEFAULT 0, wr REAL DEFAULT 0.0, pnl_avg REAL DEFAULT 0.0, created_ts REAL, scade_ts REAL, hits INTEGER DEFAULT 0, hits_saved REAL DEFAULT 0.0, last_hit_ts REAL DEFAULT 0, note TEXT DEFAULT '' ) """) conn.commit() conn.close() return True except Exception as e: logging.error(f"[CM] ensure_capsule_table: {e}") return False
